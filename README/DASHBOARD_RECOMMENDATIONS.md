## 📝 نمونه کد (Quick Start):

### داشبورد زبان‌آموز - نمای کلی:

```python
# dashboard/views.py
class StudentHomeView(LoginRequiredMixin, UserPassesTestMixin, TemplateView):
    template_name = 'dashboard/student_home.html'
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        student = self.request.user.student
        
        # آمار کلی
        context['stats'] = {
            'active_courses': student.enrollments.filter(status='confirmed').count(),
            'profile_completion': student.get_profile_completion_percentage(),
            'average_score': student.get_average_score(),
            'pending_assignments': student.get_pending_assignments_count(),
        }
        
        # کلاس بعدی
        context['next_class'] = student.get_next_class()
        
        # تکالیف نزدیک به deadline
        context['urgent_assignments'] = student.get_urgent_assignments()
        
        # پیشرفت این ماه
        context['monthly_progress'] = student.get_monthly_progress()
        
        return context
```
